cmake_minimum_required(VERSION 3.20)
project(compact_gazebo_physics LANGUAGES CXX)
find_package(gz-sim8 REQUIRED)
find_package(gz-plugin2 REQUIRED COMPONENTS register)
add_library(compact_polar SHARED PolarSystem.cpp)
target_compile_features(compact_polar PRIVATE cxx_std_20)
target_compile_options(compact_polar PRIVATE -Wall -Wextra -Wpedantic -fno-fast-math -ffp-contract=off)
target_link_libraries(compact_polar PRIVATE gz-sim8::gz-sim8 gz-plugin2::register)
