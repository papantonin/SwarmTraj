cmake_minimum_required(VERSION 3.20)
project(swarmtraj_native VERSION 0.1.0 LANGUAGES CXX)
option(SWARM_BUILD_PYTHON "Build the optional CPython buffer binding" ON)
option(SWARM_SANITIZE "Enable ASan/UBSan on standalone numerical tests" OFF)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
file(GLOB native_sources CONFIGURE_DEPENDS RELATIVE "${CMAKE_CURRENT_SOURCE_DIR}" "src/*.cpp")
set(core_sources ${native_sources})
list(FILTER core_sources EXCLUDE REGEX "(^src/python\\.cpp$|_python\\.cpp$)")
set(binding_sources ${native_sources})
list(FILTER binding_sources INCLUDE REGEX "(^src/python\\.cpp$|_python\\.cpp$)")
add_library(swarmtraj STATIC ${core_sources})
set_target_properties(swarmtraj PROPERTIES POSITION_INDEPENDENT_CODE ON)
target_include_directories(swarmtraj PUBLIC $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include> $<INSTALL_INTERFACE:include>)
if(MSVC)
  target_compile_options(swarmtraj PUBLIC /fp:strict)
else()
  target_compile_options(swarmtraj PUBLIC -fno-fast-math -ffp-contract=off)
  target_compile_options(swarmtraj PRIVATE -Wall -Wextra -Wpedantic)
  if(SWARM_SANITIZE)
    target_compile_options(swarmtraj PUBLIC -fsanitize=address,undefined -fno-omit-frame-pointer)
    target_link_options(swarmtraj PUBLIC -fsanitize=address,undefined)
  endif()
endif()
include(CTest)
if(BUILD_TESTING)
  add_executable(swarmtraj_approach_path_test tests/test_approach_path.cpp)
  target_link_libraries(swarmtraj_approach_path_test PRIVATE swarmtraj)
  add_test(NAME bounded_tangent_geometry COMMAND swarmtraj_approach_path_test)
  add_executable(swarmtraj_test tests/test_flight.cpp)
  target_link_libraries(swarmtraj_test PRIVATE swarmtraj)
  add_test(NAME numerical_core COMMAND swarmtraj_test)
  add_executable(swarmtraj_execution_test tests/test_execution.cpp)
  target_link_libraries(swarmtraj_execution_test PRIVATE swarmtraj)
  add_test(NAME local_execution COMMAND swarmtraj_execution_test)
  add_executable(swarmtraj_validation_test tests/test_validation.cpp)
  target_link_libraries(swarmtraj_validation_test PRIVATE swarmtraj)
  add_test(NAME independent_validation COMMAND swarmtraj_validation_test)
  add_executable(swarmtraj_catalog_test tests/test_catalog.cpp)
  target_link_libraries(swarmtraj_catalog_test PRIVATE swarmtraj)
  add_test(NAME deterministic_catalog COMMAND swarmtraj_catalog_test)
  add_executable(swarmtraj_search_test tests/test_search.cpp)
  target_link_libraries(swarmtraj_search_test PRIVATE swarmtraj)
  add_test(NAME bounded_joint_search COMMAND swarmtraj_search_test)
  add_executable(swarmtraj_mission_test tests/test_mission.cpp)
  target_link_libraries(swarmtraj_mission_test PRIVATE swarmtraj)
  add_test(NAME complete_mission COMMAND swarmtraj_mission_test)
endif()
if(SWARM_BUILD_PYTHON)
  find_package(Python3 3.12 COMPONENTS Interpreter Development.Module REQUIRED)
  Python3_add_library(_trajectory_native MODULE WITH_SOABI ${binding_sources})
  target_link_libraries(_trajectory_native PRIVATE swarmtraj)
  set(source_hash "")
  file(GLOB_RECURSE native_headers CONFIGURE_DEPENDS RELATIVE "${CMAKE_CURRENT_SOURCE_DIR}" "include/*.hpp" "src/*.hpp")
  set(identity_files ${native_sources} ${native_headers})
  list(SORT identity_files)
  foreach(file IN LISTS identity_files)
    file(SHA256 "${CMAKE_CURRENT_SOURCE_DIR}/${file}" digest)
    string(APPEND source_hash "${digest}")
  endforeach()
  string(SHA256 source_hash "${source_hash}")
  file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/build_info.hpp" "#define SWARM_SOURCE_SHA \"${source_hash}\"\n")
  target_include_directories(_trajectory_native PRIVATE "${CMAKE_CURRENT_BINARY_DIR}")
  if(SWARM_PYTHON_OUTPUT_DIRECTORY)
    set_target_properties(_trajectory_native PROPERTIES LIBRARY_OUTPUT_DIRECTORY "${SWARM_PYTHON_OUTPUT_DIRECTORY}")
  endif()
endif()
install(TARGETS swarmtraj ARCHIVE DESTINATION lib)
install(DIRECTORY include/ DESTINATION include)
